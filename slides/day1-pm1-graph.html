const State = Annotation.Root({
  messages: Annotation<any[]>({
    reducer: (prev, next) => [...prev, ...next], default: () => [] }),
});

async function tutor(state: typeof State.State) {
  return { messages: [await chat(state.messages, tools)] };
}

function toolsNode(state: typeof State.State) {
  return { messages: state.messages.at(-1).tool_calls.map((call) => ({
    role: "tool", tool_call_id: call.id,
    content: JSON.stringify(lookupWord(JSON.parse(call.function.arguments).word)),
  })) };
}

const route = (state) => state.messages.at(-1).tool_calls?.length ? "tools" : END;

const agent = new StateGraph(State)
  .addNode("tutor", tutor).addNode("tools", toolsNode)
  .addEdge(START, "tutor")
  .addConditionalEdges("tutor", route, ["tools", END])
  .addEdge("tools", "tutor")
  .compile();

await agent.invoke({ messages: [SYSTEM, { role: "user", content: question }] },
                   { recursionLimit: 25 });   // 상한 — 걸리면 에러로 멈춘다

}

  return state.messages.at(-1).tool_calls?.length ? "tools" : END;
