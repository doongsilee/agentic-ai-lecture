function orderNode(state: typeof State.State) {
  console.log("[NODE] ▶ order");
  const calls = state.messages.at(-1).tool_calls;
  const reply = interrupt({ request: calls.map((c: any) => `${c.function.name} ${c.function.arguments}`) });
  if (reply === "y") return runCalls(state);   // 청구는 interrupt() 뒤에
  return { messages: calls.map((c: any) => ({ role: "tool", tool_call_id: c.id,
    content: JSON.stringify({ 결과: `청구하지 않았습니다. 담당자 의견: ${reply}` }) })) };
}

while ((await graph.getState(thread1)).next.includes("order")) {

  for (const line of pending.value.request) console.log(`[멈춤] ${line}`);
  const reply = (await rl.question("청구할까요? (y / 아니면 담당자 의견) ")).trim();
  await run(new Command({ resume: reply }));
}
